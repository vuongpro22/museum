server {
    server_name binhvuong.id.vn www.binhvuong.id.vn;

    client_max_body_size 12M;

    # Frontend museum (vite build)
    root /var/www/museum/frontend/dist;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # Backend museum (Node.js server)
    location /api/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/binhvuong.id.vn/fullchain.pem; # >
    ssl_certificate_key /etc/letsencrypt/live/binhvuong.id.vn/privkey.pem; >
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

server {
    if ($host = www.binhvuong.id.vn) {
        return 301 https://$host$request_uri;
    }
    if ($host = binhvuong.id.vn) {
        return 301 https://$host$request_uri;
    }

    listen 80;
    server_name binhvuong.id.vn www.binhvuong.id.vn;
    return 404;
}
